----------    C O N F I G    ----------

-- folder holding vw_transporter.obj and the textures. Must end in /
local BASE_URL = "https://raw.githubusercontent.com/Equatox/TestAnims/main/vehicles/vw_transporter/"

local MODEL_SCALE = 16.9              -- 16.9 is about real size
local OFFSET = Vector(0, 0, 0)        -- from the Base prop or chip. The model's floor is at 0
local ANGLES = Angle(0, 0, 0)         -- the model faces +X at 0
local RENDER_BOUNDS = 300

local GLASS_ALPHA = 0.45              -- windscreen and front door windows, seen from both sides
local INTERIOR_MATERIAL = "models/debug/debugwhite"
local INTERIOR_COLOR = Color(130, 130, 130)
local TEXTURE_SIZE = 1024             -- every image is stretched to fill this
local LOAD_CPU = 0.4                  -- share of the CPU quota used while loading

-- left side sliding door: press E on it
local DOOR_OWNER_ONLY = false         -- true: only you can open it
local DOOR_USE_DISTANCE = 110
local DOOR_OUT = 7                    -- how far it first moves away from the van
local DOOR_SLIDE = 58                 -- how far it then slides back
local DOOR_TIME = 1.2                 -- seconds to fully open or close
local DOOR_SOUND = "doors/doormove2.wav"
local DOOR_BOX_MIN = Vector(-50, 39, 19) -- closed door, in model units x scale 16.9
local DOOR_BOX_MAX = Vector(13, 51, 104)

-- invisible collision, in the same units as DOOR_BOX. The cab is solid; only the rear is hollow
local COLLISION = true
local FLOOR_Z = 17.5                  -- players can step up to 18 without jumping
local WALL = 5                        -- wall, roof and door thickness
local BAY_REAR, BAY_FRONT = -128, 12  -- length of the hollow rear
local BAY_WIDTH = 52                  -- centre to the outside of the van
local ROOF_Z = 110
local CAB_PROFILE = {{12, 2}, {122, 2}, {122, 60}, {48, 106}, {12, 110}} -- side outline: x, z

-- mesh name in the OBJ = texture file. false uses INTERIOR_MATERIAL and INTERIOR_COLOR
local PARTS = {
    body = "livery.png",
    solids = "solids.png",
    interior = false,
    under = "vw_transporter_dessous.jpg",
    feuxar = "vw_transporter_feuxar.jpg",
    clignot = "vw_transporter_clignot.jpg",
    feuxred = "feuxred.jpg",
    plate = "VW_plate.jpg",
    glass = "glass.png",
    door_body = "livery.png",
    door_solids = "solids.png",
    door_interior = false,
    wheel_fl = "wheel.jpg",
    wheel_fr = "wheel.jpg",
    wheel_rl = "wheel.jpg",
    wheel_rr = "wheel.jpg",
    steering = "solids.png",
}

-- where each moving part pivots, in the same units as DOOR_BOX. fl = front left
local PIVOTS = {
    wheel_fl = Vector(85.0, 45.4, 20.5),
    wheel_fr = Vector(85.0, -45.6, 20.5),
    wheel_rl = Vector(-76.1, 45.4, 20.5),
    wheel_rr = Vector(-76.0, -45.6, 20.5),
    steering = Vector(53.2, 25.9, 69.0),
}
local STEERING_TILT = Angle(-24.2, 180, 0) -- points the steering wheel's local +X down its column

----------    S E R V E R    ----------

if SERVER then
    wire.adjustPorts({Base = "entity"})
    local base = nil
    local doorOpen = false
    local nextUse = 0

    local function vehicle()
        return isValid(base) and base or chip()
    end

    local function box(x1, y1, z1, x2, y2, z2)
        return {
            Vector(x1, y1, z1), Vector(x2, y1, z1), Vector(x2, y2, z1), Vector(x1, y2, z1),
            Vector(x1, y1, z2), Vector(x2, y1, z2), Vector(x2, y2, z2), Vector(x1, y2, z2),
        }
    end

    local function bodyHulls()
        local w = BAY_WIDTH
        local top = ROOF_Z - WALL
        local cab = {}
        for _, point in ipairs(CAB_PROFILE) do
            cab[#cab + 1] = Vector(point[1], -w, point[2])
            cab[#cab + 1] = Vector(point[1], w, point[2])
        end
        return {
            box(BAY_REAR, -w, 2, BAY_FRONT, w, FLOOR_Z),               -- floor and underside
            box(BAY_REAR, -w, top, BAY_FRONT, w, ROOF_Z),              -- roof
            box(BAY_REAR, -w, FLOOR_Z, BAY_REAR + WALL, w, top),       -- rear
            box(BAY_REAR, -w, FLOOR_Z, BAY_FRONT, -w + WALL, top),     -- right side
            box(BAY_REAR, w - WALL, FLOOR_Z, DOOR_BOX_MIN[1], w, top), -- left, behind the door
            cab,
        }
    end

    local function doorHull()
        return {box(DOOR_BOX_MIN[1], BAY_WIDTH - WALL, FLOOR_Z, DOOR_BOX_MAX[1], BAY_WIDTH, ROOF_Z - WALL)}
    end

    local collider = {}

    local function clearCollision()
        for _, ent in pairs(collider) do
            if isValid(ent) then ent:remove() end
        end
        collider = {}
    end

    local function spawnCollider(hulls)
        local van = vehicle()
        local pos, ang = localToWorld(OFFSET, ANGLES, van:getPos(), van:getAngles())
        local ent = prop.createCustom(pos, ang, hulls, true)
        ent:setNoDraw(true)
        if isValid(base) then
            constraint.weld(ent, base, 0, 0, 0, true)
            ent:setFrozen(false)
        end
        return ent
    end

    -- the door's collision stays put and just stops blocking players while open
    local function playerInDoorway()
        local van = vehicle()
        local pos, ang = localToWorld(OFFSET, ANGLES, van:getPos(), van:getAngles())
        local a = localToWorld(DOOR_BOX_MIN, Angle(), pos, ang)
        local b = localToWorld(DOOR_BOX_MAX, Angle(), pos, ang)
        local boxMin = Vector(math.min(a[1], b[1]), math.min(a[2], b[2]), math.min(a[3], b[3]))
        local boxMax = Vector(math.max(a[1], b[1]), math.max(a[2], b[2]), math.max(a[3], b[3]))
        return #find.inBox(boxMin, boxMax, function(ent) return ent:isPlayer() end) > 0
    end

    local function updateDoorCollision()
        local ent = collider.door
        if not isValid(ent) then return end
        timer.remove("vwDoorClear")
        if doorOpen then
            ent:setCollisionGroup(COLLISION_GROUP.DEBRIS)
        elseif playerInDoorway() then
            -- closing on someone would trap them, so wait until the doorway is clear
            ent:setCollisionGroup(COLLISION_GROUP.DEBRIS)
            timer.create("vwDoorClear", 0.25, 1, updateDoorCollision)
        else
            ent:setCollisionGroup(COLLISION_GROUP.NONE)
        end
    end

    local function buildCollision()
        if not COLLISION then return end
        clearCollision()
        hook.add("think", "vwBuildCollision", function()
            if not prop.canSpawn() then return end
            if not collider.body then
                collider.body = spawnCollider(bodyHulls())
            else
                collider.door = spawnCollider(doorHull())
                hook.remove("think", "vwBuildCollision")
                updateDoorCollision()
            end
        end)
    end

    local function sendBase(target)
        net.start("vwBase")
        net.writeEntity(vehicle())
        net.send(target)
    end

    local function sendDoor(target)
        net.start("vwDoor")
        net.writeBool(doorOpen)
        net.send(target)
    end

    hook.add("Input", "vwBaseInput", function(name, value)
        if name == "Base" then
            base = value
            sendBase()
            buildCollision()
        end
    end)

    hook.add("ClientInitialized", "vwSync", function(ply)
        sendBase(ply)
        sendDoor(ply)
    end)

    net.receive("vwUseDoor", function(_, ply)
        if DOOR_OWNER_ONLY and ply ~= owner() then return end
        if timer.curtime() < nextUse then return end
        if ply:getPos():getDistance(vehicle():getPos()) > DOOR_USE_DISTANCE + 300 then return end
        nextUse = timer.curtime() + DOOR_TIME
        doorOpen = not doorOpen
        sendDoor()
        updateDoorCollision()
        pcall(function() vehicle():emitSound(DOOR_SOUND) end)
    end)

    buildCollision()

    return
end

----------    C L I E N T    ----------

local parent = chip()
local holos = {}
local doorHolos = {}
local materials = {}
local door = {open = false, t = 0}

local function ease(x)
    x = math.clamp(x, 0, 1)
    return x * x * (3 - 2 * x)
end

-- out from the van first, then back along the rail
local function doorOffset(t)
    return Vector(-DOOR_SLIDE * ease((t - 0.25) / 0.75), DOOR_OUT * ease(t / 0.25), 0)
end

local function placeDoor()
    local localPos = OFFSET + doorOffset(door.t):getRotated(ANGLES)
    for _, holo in ipairs(doorHolos) do
        holo:setLocalPos(localPos)
    end
end

local parts = {}  -- pivoting parts by name

local function toVan(localAng)
    local _, ang = localToWorld(Vector(), localAng, Vector(), ANGLES)
    return ang
end

-- spin: rolling in degrees. steer: turned left in degrees, for the front wheels
local function setWheel(name, spin, steer)
    local holo = parts[name]
    if holo then holo:setLocalAngles(toVan(Angle(spin, steer or 0, 0))) end
end

-- degrees around the steering column
local function setSteeringWheel(degrees)
    local holo = parts.steering
    if holo then holo:setLocalAngles(toVan(Angle(STEERING_TILT[1], STEERING_TILT[2], STEERING_TILT[3] + degrees))) end
end

local function placeParts()
    for name, holo in pairs(parts) do
        holo:setLocalPos(OFFSET + PIVOTS[name]:getRotated(ANGLES))
    end
    for _, name in ipairs({"wheel_fl", "wheel_fr", "wheel_rl", "wheel_rr"}) do
        setWheel(name, 0, 0)
    end
    setSteeringWheel(0)
end

local function attach()
    if not isValid(parent) then return end
    local pos, ang = localToWorld(OFFSET, ANGLES, parent:getPos(), parent:getAngles())
    for _, holo in ipairs(holos) do
        holo:setParent()
        holo:setPos(pos)
        holo:setAngles(ang)
        holo:setParent(parent)
    end
    placeDoor()
    placeParts()
end

net.receive("vwBase", function()
    net.readEntity(function(ent)
        if isValid(ent) then
            parent = ent
            attach()
        end
    end)
end)

net.receive("vwDoor", function()
    door.open = net.readBool()
end)

hook.add("think", "vwDoorMove", function()
    local goal = door.open and 1 or 0
    if door.t == goal then return end
    door.t = math.approach(door.t, goal, timer.frametime() / DOOR_TIME)
    placeDoor()
end)

-- slab test against the door box, in the van's own space
local function lookingAtDoor(eye, dir)
    local boxMin = DOOR_BOX_MIN + doorOffset(door.t)
    local boxMax = DOOR_BOX_MAX + doorOffset(door.t)
    local near, far = 0, DOOR_USE_DISTANCE
    for axis = 1, 3 do
        if math.abs(dir[axis]) < 0.0001 then
            if eye[axis] < boxMin[axis] or eye[axis] > boxMax[axis] then return false end
        else
            local a = (boxMin[axis] - eye[axis]) / dir[axis]
            local b = (boxMax[axis] - eye[axis]) / dir[axis]
            if a > b then a, b = b, a end
            near, far = math.max(near, a), math.min(far, b)
            if near > far then return false end
        end
    end
    return true
end

hook.add("inputPressed", "vwDoorUse", function(key)
    if key ~= KEY.E or not isValid(parent) then return end
    local me = player()
    if me:isTyping() or me:inVehicle() or (DOOR_OWNER_ONLY and me ~= owner()) then return end

    local vanPos, vanAng = localToWorld(OFFSET, ANGLES, parent:getPos(), parent:getAngles())
    local eye = worldToLocal(me:getEyePos(), Angle(), vanPos, vanAng)
    local aim = worldToLocal(me:getEyePos() + me:getEyeAngles():getForward(), Angle(), vanPos, vanAng) - eye
    if lookingAtDoor(eye, aim) then
        net.start("vwUseDoor")
        net.send()
    end
end)

-- Starfall draws an image at its own size in the corner of the texture, so fill it
local function fillTexture(mat, url, w, h, layout)
    if layout then layout(0, 0, TEXTURE_SIZE, TEXTURE_SIZE) end
end

local function makeMaterial(name, file)
    if file == false then return material.load(INTERIOR_MATERIAL) end
    if materials[file] then return materials[file] end
    local mat = material.create("VertexLitGeneric")
    mat:setTextureURL("$basetexture", BASE_URL .. file, fillTexture)
    if name == "glass" then
        mat:setInt("$flags", 2097152 + 8192) -- translucent, drawn from both sides
        mat:setFloat("$alpha", GLASS_ALPHA)
    end
    materials[file] = mat
    return mat
end

local function build(meshes)
    for name, file in pairs(PARTS) do
        local partMesh = meshes[name]
        if partMesh then
            local holo = holograms.create(chip():getPos(), Angle(), "models/hunter/blocks/cube025x025x025.mdl", Vector(MODEL_SCALE))
            holo:setMesh(partMesh)
            holo:setMeshMaterial(makeMaterial(name, file))
            holo:setRenderBounds(Vector(-RENDER_BOUNDS), Vector(RENDER_BOUNDS))
            if file == false then holo:setColor(INTERIOR_COLOR) end
            if name == "glass" then holo:setRenderGroup(RENDERGROUP.TRANSLUCENT) end
            holos[#holos + 1] = holo
            if name:sub(1, 5) == "door_" then doorHolos[#doorHolos + 1] = holo end
            if PIVOTS[name] then parts[name] = holo end
        else
            printConsole("VW Transporter: OBJ has no part called " .. name)
        end
    end
    attach()
end

http.get(BASE_URL .. "vw_transporter.obj", function(body)
    local meshes = nil
    local load = coroutine.wrap(function()
        meshes = mesh.createFromObj(body, true)
        return true
    end)

    hook.add("think", "vwLoad", function()
        while cpuAverage() < cpuMax() * LOAD_CPU do
            if load() then
                hook.remove("think", "vwLoad")
                build(meshes)
                return
            end
        end
    end)
end, function(reason)
    printConsole("VW Transporter: download failed (" .. tostring(reason) .. ")")
end)
